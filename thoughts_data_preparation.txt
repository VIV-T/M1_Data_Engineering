import re

# Example scene text
scene_text = "James Cole (Bruce Willis) is strapped to a bed, looking disoriented. He asks, 'Where am I? What's happening?' Staff members glance up at the air vent above."

# Define keywords or phrases for tropes
tropes = {
    "Air Vent Passageway": ["air vent", "escape", "ventilation shaft"],
    "Cassandra Truth": ["where am I?", "what's happening?", "no one believes me"],
    "Cuckoo Nest": ["hospital", "strapped down", "mental institution"]
}

# Function to label tropes
def label_tropes(scene_text):
    scene_tropes = []
    for trope, keywords in tropes.items():
        for keyword in keywords:
            if re.search(r'\b' + re.escape(keyword) + r'\b', scene_text, re.IGNORECASE):
                scene_tropes.append(trope)
                break  # Stop checking once a match is found
    return scene_tropes

# Apply the function to the scene
labeled_tropes = label_tropes(scene_text)
print(labeled_tropes)  # Output: ['Air Vent Passageway', 'Cassandra Truth']

import re

# Example list of tropes with descriptions
tropes = {
    "Air Vent Passageway": "A trope where a character escapes or tries to escape through a ventilation shaft or air duct.",
    "Cassandra Truth": "A trope where a character knows the truth, but no one believes them.",
    "Chewing the Scenery": "An actor or character overacts or delivers lines in a hyper-dramatic way, often unintentionally comedic.",
    "Cuckoo Nest": "A trope where a character is institutionalized or trapped in a psychiatric institution."
}

# Example scene text
scene_text = "James Cole (Bruce Willis) is strapped to a bed, looking disoriented. He asks, 'Where am I? What's happening?' Staff members glance up at the air vent above."

# Function to extract keywords from trope descriptions
def extract_keywords(description):
    # Simple method to extract important keywords from the description
    # You can improve this by using NLP techniques to find the most relevant words
    # This example uses regex to find possible keywords (nouns or verbs)
    return re.findall(r'\b\w+\b', description.lower())

# Map each trope to its keywords
trope_keywords = {trope: extract_keywords(description) for trope, description in tropes.items()}

# Function to match scene text with tropes based on keywords
def label_tropes(scene_text, trope_keywords):
    scene_tropes = []
    for trope, keywords in trope_keywords.items():
        for keyword in keywords:
            if re.search(r'\b' + re.escape(keyword) + r'\b', scene_text.lower(), re.IGNORECASE):

                break  # Stop checking after the first match for efficiency
    return scene_tropes

# Apply the function to the scene text
